# All source downloads stay in this workspace. Never replace an existing checkout.
function(lm_dependency name url ref revision)
    set(dir "${CMAKE_SOURCE_DIR}/.deps/source/${name}")
    if(NOT EXISTS "${dir}/.git")
        execute_process(COMMAND "${GIT_EXECUTABLE}" clone --depth 1 --branch "${ref}" "${url}" "${dir}" RESULT_VARIABLE result)
        if(NOT result EQUAL 0)
            message(FATAL_ERROR "Could not download ${name}")
        endif()
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${dir}" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE result)
    if(NOT result EQUAL 0 OR NOT actual STREQUAL revision)
        message(FATAL_ERROR "${name} must be at ${revision}; found ${actual}. Preserve any local changes before checking out the pinned revision.")
    endif()
endfunction()
